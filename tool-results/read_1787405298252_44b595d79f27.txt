     1→'use client';
     2→
     3→import { useEffect, useState, useCallback } from 'react';
     4→import {

    14→/* ================================================================== */
    15→/*  Domain Types                                                       */

    17→
    18→interface UserProfile {
    19→  pin: string;
    20→  fullName: string;
    21→  phone: string | null;
    22→  avatarUrl: string | null;
    23→}
    24→
    25→interface Balance {
    26→  amount: number;
    27→  currency: string;
    28→  paidUntil: string;
    29→  isPaid: boolean;
    30→  paidUntilLabel: string;
    31→}
    32→
    33→interface ActiveService {
    34→  id: string;
    35→  name: string;
    36→  category: string;
    37→  cost: number;
    38→  status: string;
    39→  warningMessage: string | null;
    40→  billingCycle: string | null;
    41→}
    42→
    43→interface ProfileResponse {
    44→  user: UserProfile;
    45→  balance: Balance;
    46→  activeServices: ActiveService[];
    47→}
    48→
    49→interface Transaction {
    50→  id: string;
    51→  type: string;
    52→  amount: number;
    53→  description: string;
    54→  date: string;
    55→  status: string;
    56→}
    57→
    58→interface NewsItem {
    59→  id: string;
    60→  title: string;
    61→  summary: string;
    62→  imageUrl: string | null;
    63→  publishedAt: string;
    64→  readCount: number | null;
    65→}
    66→
    67→type FetchState = 'idle' | 'loading' | 'success' | 'error';
    68→
    69→/* ================================================================== */
    70→/*  Utility                                                            */
    71→/* ================================================================== */
    72→
    73→function formatDate(iso: string): string {
    74→  const d = new Date(iso);
    75→  return d.toLocaleDateString('ru-RU', { day: 'numeric', month: 'long', year: 'numeric' });
    76→}
    77→
    78→function txIcon(type: string) {
    79→  switch (type) {
    80→    case 'topUp':
    81→    case 'bonus':
    82→      return <ArrowDownLeft className="h-5 w-5 text-emerald-500" />;
    83→    case 'refund':
    84→      return <RefreshCw className="h-5 w-5 text-blue-500" />;
    85→    default:
    86→      return <ArrowUpRight className="h-5 w-5 text-red-400" />;
    87→  }
    88→}
    89→
    90→function txSign(type: string): string {
    91→  return type === 'topUp' || type === 'bonus' || type === 'refund' ? '+' : '-';
    92→}
    93→
    94→function txColor(type: string): string {
    95→  return type === 'topUp' || type === 'bonus' || type === 'refund' ? 'text-emerald-600' : 'text-red-500';
    96→}
    97→
    98→function txLabel(type: string): string {
    99→  switch (type) {
   100→    case 'topUp': return 'Пополнение';
   101→    case 'payment': return 'Оплата';
   102→    case 'refund': return 'Возврат';
   103→    case 'bonus': return 'Бонус';
   104→    default: return type;
   105→  }
   106→}

   111→
   112→export default function DashboardScreen() {
   113→  /* ---- Profile state ---- */
   114→  const [profile, setProfile] = useState<ProfileResponse | null>(null);
   115→  const [profileState, setProfileState] = useState<FetchState>('idle');
   116→
   117→  /* ---- Transactions state ---- */
   118→  const [transactions, setTransactions] = useState<Transaction[]>([]);
   119→  const [txState, setTxState] = useState<FetchState>('idle');
   120→
   121→  /* ---- News state ---- */
   122→  const [news, setNews] = useState<NewsItem[]>([]);
   123→  const [newsState, setNewsState] = useState<FetchState>('idle');
   124→
   125→  /* ---- UI state ---- */
   126→  const [activeTab, setActiveTab] = useState(0);
   127→  const [topUpOpen, setTopUpOpen] = useState(false);
   128→  const [historyOpen, setHistoryOpen] = useState(false);
   129→  const [topUpAmount, setTopUpAmount] = useState('');
   130→  const [topUpLoading, setTopUpLoading] = useState(false);
   131→  const [selectedNews, setSelectedNews] = useState<NewsItem | null>(null);
   132→
   133→  /* ---- Fetchers ---- */
   134→  const fetchProfile = useCallback(async () => {
   135→    setProfileState('loading');
   136→    try {
   137→      const res = await fetch('/api/account/profile');
   138→      if (!res.ok) throw new Error();
   139→      setProfile(await res.json());
   140→      setProfileState('success');
   141→    } catch { setProfileState('error'); }
   142→  }, []);
   143→
   144→  const fetchTransactions = useCallback(async () => {
   145→    setTxState('loading');
   146→    try {
   147→      const res = await fetch('/api/transactions');
   148→      if (!res.ok) throw new Error();
   149→      const json = await res.json();
   150→      setTransactions(json.transactions);
   151→      setTxState('success');
   152→    } catch { setTxState('error'); }
   153→  }, []);
   154→
   155→  const fetchNews = useCallback(async () => {
   156→    setNewsState('loading');
   157→    try {
   158→      const res = await fetch('/api/news');
   159→      if (!res.ok) throw new Error();
   160→      const json = await res.json();
   161→      setNews(json.news);
   162→      setNewsState('success');
   163→    } catch { setNewsState('error'); }
   164→  }, []);
   165→
   166→  /* ---- Load data on tab switch ---- */
   167→  useEffect(() => { fetchProfile(); }, [fetchProfile]);
   168→  useEffect(() => { if (activeTab === 1) fetchTransactions(); }, [activeTab, fetchTransactions]);
   169→  useEffect(() => { if (activeTab === 2) fetchNews(); }, [activeTab, fetchNews]);
   170→
   171→  /* ---- Top-up handler ---- */
   172→  const handleTopUp = async () => {
   173→    const amount = parseFloat(topUpAmount);
   174→    if (isNaN(amount) || amount <= 0) {
   175→      toast.error('Введите корректную сумму');
   176→      return;
   177→    }
   178→    setTopUpLoading(true);
   179→    try {
   180→      const res = await fetch('/api/top-up', {
   181→        method: 'POST',
   182→        headers: { 'Content-Type': 'application/json' },
   183→        body: JSON.stringify({ amount }),
   184→      });
   185→      if (!res.ok) throw new Error();
   186→      const json = await res.json();
   187→      toast.success(`Баланс пополнен на ${amount} ₽`);
   188→      setTopUpAmount('');
   189→      setTopUpOpen(false);
   190→      fetchProfile();
   191→    } catch {
   192→      toast.error('Ошибка пополнения');
   193→    } finally {
   194→      setTopUpLoading(false);
   195→    }
   196→  };
   197→
   198→  /* ---- Tabs ---- */
   199→  const tabs = [
   200→    { icon: Home, label: 'Главная' },
   201→    { icon: CreditCard, label: 'Оплата' },
   202→    { icon: FileText, label: 'Новости' },
   203→    { icon: HelpCircle, label: 'Поддержка' },
   204→  ];
   205→
   206→  /* ---- Render ---- */
   207→  return (
   208→    <div className="relative mx-auto min-h-dvh w-full max-w-[430px] bg-white font-sans">
   209→      {/* ============ CONTENT AREA ============ */}
   210→      {activeTab === 0 && (
   211→        <HomeTab
   212→          profile={profile} profileState={profileState}
   213→          onTopUp={() => setTopUpOpen(true)}
   214→          onHistory={() => { setHistoryOpen(true); fetchTransactions(); }}
   215→          onRetry={fetchProfile}
   216→        />
   217→      )}
   218→      {activeTab === 1 && <PaymentTab transactions={transactions} txState={txState} onRetry={fetchTransactions} />}
   219→      {activeTab === 2 && <NewsTab news={news} newsState={newsState} onRetry={fetchNews} onSelect={setSelectedNews} />}
   220→      {activeTab === 3 && <SupportTab />}
   221→
   222→      {/* ============ TOP-UP SHEET ============ */}
   223→      <Sheet open={topUpOpen} onOpenChange={setTopUpOpen}>
   224→        <SheetContent side="bottom" className="mx-auto max-w-[430px] rounded-t-2xl px-6 pb-8 pt-4">
   225→          <SheetHeader>
   226→            <SheetTitle className="text-xl font-bold text-gray-900">Пополнить баланс</SheetTitle>
   227→            <SheetDescription className="text-gray-500">Выберите или введите сумму пополнения</SheetDescription>
   228→          </SheetHeader>
   229→          <div className="mt-4">
   230→            {/* Quick amounts */}
   231→            <div className="grid grid-cols-3 gap-3">
   232→              {[100, 200, 500, 1000, 2000, 5000].map((a) => (
   233→                <button
   234→                  key={a}
   235→                  type="button"
   236→                  onClick={() => setTopUpAmount(String(a))}
   237→                  className={`rounded-xl border-2 py-3 text-center text-lg font-semibold transition-all ${
   238→                    topUpAmount === String(a)
   239→                      ? 'border-orange-500 bg-orange-50 text-orange-600'
   240→                      : 'border-gray-200 bg-white text-gray-700 hover:border-orange-300'
   241→                  }`}
   242→                >
   243→                  {a} ₽
   244→                </button>
   245→              ))}
   246→            </div>
   247→            {/* Custom input */}
   248→            <div className="mt-4">
   249→              <label htmlFor="topup-input" className="mb-1.5 block text-sm font-medium text-gray-600">
   250→                Другая сумма
   251→              </label>
   252→              <input
   253→                id="topup-input"
   254→                type="number"
   255→                inputMode="numeric"
   256→                placeholder="0"
   257→                value={topUpAmount}
   258→                onChange={(e) => setTopUpAmount(e.target.value)}
   259→                className="w-full rounded-xl border-2 border-gray-200 px-4 py-3 text-xl font-semibold text-gray-900 outline-none transition-colors focus:border-orange-500"
   260→              />
   261→            </div>
   262→            {/* Submit */}
   263→            <button
   264→              type="button"
   265→              onClick={handleTopUp}
   266→              disabled={topUpLoading || !topUpAmount}
   267→              className="mt-5 flex w-full items-center justify-center gap-2 rounded-xl bg-orange-500 px-6 py-3.5 text-lg font-semibold text-white transition-colors hover:bg-orange-600 disabled:opacity-50 disabled:cursor-not-allowed"
   268→            >
   269→              {topUpLoading ? <Loader2 className="h-5 w-5 animate-spin" /> : <Wallet className="h-5 w-5" />}
   270→              Пополнить
   271→            </button>
   272→          </div>
   273→        </SheetContent>
   274→      </Sheet>
   275→
   276→      {/* ============ HISTORY SHEET ============ */}
   277→      <Sheet open={historyOpen} onOpenChange={setHistoryOpen}>
   278→        <SheetContent side="bottom" className="mx-auto max-w-[430px] rounded-t-2xl px-6 pb-8 pt-4">
   279→          <SheetHeader>
   280→            <SheetTitle className="text-xl font-bold text-gray-900">История операций</SheetTitle>
   281→            <SheetDescription className="text-gray-500">Все транзакции по вашему аккаунту</SheetDescription>
   282→          </SheetHeader>
   283→          <div className="mt-2 max-h-[60vh] overflow-y-auto">
   284→            <TransactionList transactions={transactions} state={txState} onRetry={fetchTransactions} />
   285→          </div>
   286→        </SheetContent>
   287→      </Sheet>
   288→
   289→      {/* ============ NEWS DETAIL SHEET ============ */}
   290→      <Sheet open={!!selectedNews} onOpenChange={() => setSelectedNews(null)}>
   291→        <SheetContent side="bottom" className="mx-auto max-w-[430px] rounded-t-2xl px-6 pb-8 pt-4">
   292→          {selectedNews && (
   293→            <>
   294→              <SheetHeader>
   295→                <SheetTitle className="text-xl font-bold text-gray-900 leading-snug">{selectedNews.title}</SheetTitle>
   296→                <SheetDescription className="text-gray-400 text-sm">
   297→                  <Clock className="mr-1 inline h-3.5 w-3.5" />
   298→                  {formatDate(selectedNews.publishedAt)}
   299→                  {selectedNews.readCount != null && ` · ${selectedNews.readCount} просмотров`}
   300→                </SheetDescription>
   301→              </SheetHeader>
   302→              <p className="mt-4 text-[16px] leading-relaxed text-gray-700">{selectedNews.summary}</p>
   303→            </>
   304→          )}
   305→        </SheetContent>
   306→      </Sheet>
   307→
   308→      {/* ============ BOTTOM NAV ============ */}
   309→      <nav className="fixed bottom-0 left-1/2 z-50 w-full max-w-[430px] -translate-x-1/2 border-t border-gray-200 bg-white shadow-[0_-2px_10px_rgba(0,0,0,0.06)]">
   310→        <ul className="flex justify-around py-3 pb-5">
   311→          {tabs.map((tab, idx) => {
   312→            const Icon = tab.icon;
   313→            const active = idx === activeTab;
   314→            return (
   315→              <li key={tab.label}>
   316→                <button
   317→                  type="button"
   318→                  onClick={() => setActiveTab(idx)}
   319→                  className={`flex flex-col items-center gap-1.5 transition-colors ${
   320→                    active ? 'text-orange-500' : 'text-gray-400 hover:text-gray-600'
   321→                  }`}
   322→                >
   323→                  <Icon className="h-[26px] w-[26px]" strokeWidth={active ? 2.2 : 1.8} />
   324→                  <span className="text-[13px] font-medium">{tab.label}</span>
   325→                </button>
   326→              </li>
   327→            );
   328→          })}
   329→        </ul>
   330→      </nav>
   331→    </div>
   332→  );
   333→}

   496→function NewsTab({
   497→  news, newsState, onRetry, onSelect,
   498→}: {
   499→  news: NewsItem[];
   500→  newsState: FetchState;
   501→  onRetry: () => void;
   502→  onSelect: (n: NewsItem) => void;
   503→}) {
   504→  return (
   505→    <div className="min-h-dvh bg-gray-50 px-5 pt-6 pb-28">

   559→function SupportTab() {

   688→function TransactionList({
   689→  transactions, state, onRetry,
   690→}: {
   691→  transactions: Transaction[];
   692→  state: FetchState;
   693→  onRetry: () => void;
   694→}) {
   695→  if (state === 'loading') {
   696→    return (
   697→      <div className="flex items-center justify-center py-10">
   698→        <Loader2 className="h-6 w-6 animate-spin text-gray-300" />
   699→      </div>
   700→    );
   701→  }
   702→  if (state === 'error') {
   703→    return (
   704→      <div className="py-10 text-center">
   705→        <p className="mb-2 text-sm text-gray-400">Ошибка загрузки</p>
   706→        <button type="button" onClick={onRetry} className="text-sm text-orange-500 underline">Повторить</button>
   707→      </div>
   708→    );
   709→  }
   710→  if (transactions.length === 0) {
   711→    return (
   712→      <div className="py-10 text-center">
   713→        <MessageSquare className="mx-auto mb-2 h-8 w-8 text-gray-300" />
   714→        <p className="text-sm text-gray-400">Нет транзакций</p>
   715→      </div>
   716→    );
   717→  }
   718→  return (
   719→    <div className="divide-y divide-gray-100">
   720→      {transactions.map((tx) => (
   721→        <div key={tx.id} className="flex items-center gap-3 py-3 first:pt-0 last:pb-0">
   722→          <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-gray-50">
   723→            {txIcon(tx.type)}
   724→          </div>
   725→          <div className="min-w-0 flex-1">
   726→            <p className="truncate text-[14px] font-medium text-gray-800">{tx.description}</p>
   727→            <div className="flex items-center gap-2 text-[12px] text-gray-400">
   728→              <span className="rounded bg-gray-100 px-1.5 py-0.5 text-[11px] font-medium text-gray-500">
   729→                {txLabel(tx.type)}
   730→              </span>
   731→              <span>{formatDate(tx.date)}</span>
   732→            </div>
   733→          </div>
   734→          <span className={`shrink-0 text-[15px] font-semibold tabular-nums ${txColor(tx.type)}`}>
   735→            {txSign(tx.type)}{tx.amount.toFixed(0)} ₽
   736→          </span>
   737→        </div>
   738→      ))}

   740→  );
