   116→   116→function loadAuth(): AuthState {
   117→   117→  if (typeof window === 'undefined') return { token: null, fullName: null };
   118→   118→  try {
   119→   119→    const raw = localStorage.getItem(AUTH_KEY);
   120→   120→    return raw ? JSON.parse(raw) : { token: null, fullName: null };
   121→   121→  } catch {
   122→   122→    return { token: null, fullName: null };
   123→   123→  }
   124→   124→}

   131→   131→  localStorage.removeItem(AUTH_KEY);
