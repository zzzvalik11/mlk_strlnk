    73→    73→function formatDate(iso: string): string {
    74→    74→  const d = new Date(iso);
    75→    75→  return d.toLocaleDateString('ru-RU', { day: 'numeric', month: 'long', year: 'numeric' });
    76→    76→}

    78→    78→function txIcon(type: string) {
    79→    79→  switch (type) {
    80→    80→    case 'topUp':
    81→    81→    case 'bonus':
    82→    82→      return <ArrowDownLeft className="h-5 w-5 text-emerald-500" />;
    83→    83→    case 'refund':
    84→    84→      return <RefreshCw className="h-5 w-5 text-blue-500" />;
    85→    85→    default:
    86→    86→      return <ArrowUpRight className="h-5 w-5 text-red-400" />;
    87→    87→  }
    88→    88→}
    89→    89→
    90→    90→function txSign(type: string): string {
    91→    91→  return type === 'topUp' || type === 'bonus' || type === 'refund' ? '+' : '-';
    92→    92→}
    93→    93→
    94→    94→function txColor(type: string): string {
    95→    95→  return type === 'topUp' || type === 'bonus' || type === 'refund' ? 'text-emerald-600' : 'text-red-500';
    96→    96→}
    97→    97→
    98→    98→function txLabel(type: string): string {
    99→    99→  switch (type) {
   100→   100→    case 'topUp': return 'Пополнение';
   101→   101→    case 'payment': return 'Оплата';
   102→   102→    case 'refund': return 'Возврат';
   103→   103→    case 'bonus': return 'Бонус';
   104→   104→    default: return type;
   105→   105→  }
   106→   106→}

   111→   111→
   112→   112→export default function DashboardScreen() {
   113→   113→  /* ---- Profile state ---- */
   114→   114→  const [profile, setProfile] = useState<ProfileResponse | null>(null);
   115→   115→  const [profileState, setProfileState] = useState<FetchState>('idle');
   116→   116→
   117→   117→  /* ---- Transactions state ---- */
   118→   118→  const [transactions, setTransactions] = useState<Transaction[]>([]);
   119→   119→  const [txState, setTxState] = useState<FetchState>('idle');
   120→   120→
   121→   121→  /* ---- News state ---- */
   122→   122→  const [news, setNews] = useState<NewsItem[]>([]);
   123→   123→  const [newsState, setNewsState] = useState<FetchState>('idle');
   124→   124→
   125→   125→  /* ---- UI state ---- */
   126→   126→  const [activeTab, setActiveTab] = useState(0);
   127→   127→  const [topUpOpen, setTopUpOpen] = useState(false);
   128→   128→  const [historyOpen, setHistoryOpen] = useState(false);
   129→   129→  const [topUpAmount, setTopUpAmount] = useState('');

   131→   131→  const [selectedNews, setSelectedNews] = useState<NewsItem | null>(null);
   132→   132→
   133→   133→  /* ---- Fetchers ---- */
   134→   134→  const fetchProfile = useCallback(async () => {
   135→   135→    setProfileState('loading');
   136→   136→    try {
   137→   137→      const res = await fetch('/api/account/profile');
   138→   138→      if (!res.ok) throw new Error();
   139→   139→      setProfile(await res.json());
   140→   140→      setProfileState('success');
   141→   141→    } catch { setProfileState('error'); }
   142→   142→  }, []);
   143→   143→
   144→   144→  const fetchTransactions = useCallback(async () => {
   145→   145→    setTxState('loading');
   146→   146→    try {
   147→   147→      const res = await fetch('/api/transactions');
   148→   148→      if (!res.ok) throw new Error();
   149→   149→      const json = await res.json();
   150→   150→      setTransactions(json.transactions);
   151→   151→      setTxState('success');
   152→   152→    } catch { setTxState('error'); }
   153→   153→  }, []);
   154→   154→
   155→   155→  const fetchNews = useCallback(async () => {
   156→   156→    setNewsState('loading');
   157→   157→    try {
   158→   158→      const res = await fetch('/api/news');
   159→   159→      if (!res.ok) throw new Error();
   160→   160→      const json = await res.json();
   161→   161→      setNews(json.news);
   162→   162→      setNewsState('success');
   163→   163→    } catch { setNewsState('error'); }
   164→   164→  }, []);
   165→   165→
   166→   166→  /* ---- Load data on tab switch ---- */
   167→   167→  useEffect(() => { fetchProfile(); }, [fetchProfile]);
   168→   168→  useEffect(() => { if (activeTab === 1) fetchTransactions(); }, [activeTab, fetchTransactions]);
   169→   169→  useEffect(() => { if (activeTab === 2) fetchNews(); }, [activeTab, fetchNews]);
   170→   170→
   171→   171→  /* ---- Top-up handler ---- */

   173→   173→    const amount = parseFloat(topUpAmount);
   174→   174→    if (isNaN(amount) || amount <= 0) {
   175→   175→      toast.error('Введите корректную сумму');
   176→   176→      return;
   177→   177→    }
   178→   178→    setTopUpLoading(true);
   179→   179→    try {
   180→   180→      const res = await fetch('/api/top-up', {
   181→   181→        method: 'POST',
   182→   182→        headers: { 'Content-Type': 'application/json' },
   183→   183→        body: JSON.stringify({ amount }),
   184→   184→      });
   185→   185→      if (!res.ok) throw new Error();
   186→   186→      const json = await res.json();

   188→   188→      setTopUpAmount('');
   189→   189→      setTopUpOpen(false);
   190→   190→      fetchProfile();
   191→   191→    } catch {
   192→   192→      toast.error('Ошибка пополнения');
   193→   193→    } finally {
   194→   194→      setTopUpLoading(false);
   195→   195→    }
   196→   196→  };
   197→   197→
   198→   198→  /* ---- Tabs ---- */
   199→   199→  const tabs = [
   200→   200→    { icon: Home, label: 'Главная' },
   201→   201→    { icon: CreditCard, label: 'Оплата' },
   202→   202→    { icon: FileText, label: 'Новости' },
   203→   203→    { icon: HelpCircle, label: 'Поддержка' },
   204→   204→  ];
   205→   205→
   206→   206→  /* ---- Render ---- */
   207→   207→  return (
   208→   208→    <div className="relative mx-auto min-h-dvh w-full max-w-[430px] bg-white font-sans">
   209→   209→      {/* ============ CONTENT AREA ============ */}
   210→   210→      {activeTab === 0 && (
   211→   211→        <HomeTab
   212→   212→          profile={profile} profileState={profileState}
   213→   213→          onTopUp={() => setTopUpOpen(true)}
   214→   214→          onHistory={() => { setHistoryOpen(true); fetchTransactions(); }}
   215→   215→          onRetry={fetchProfile}
   216→   216→        />
   217→   217→      )}
   218→   218→      {activeTab === 1 && <PaymentTab transactions={transactions} txState={txState} onRetry={fetchTransactions} />}
   219→   219→      {activeTab === 2 && <NewsTab news={news} newsState={newsState} onRetry={fetchNews} onSelect={setSelectedNews} />}
   220→   220→      {activeTab === 3 && <SupportTab />}
   221→   221→
   222→   222→      {/* ============ TOP-UP SHEET ============ */}
   223→   223→      <Sheet open={topUpOpen} onOpenChange={setTopUpOpen}>
   224→   224→        <SheetContent side="bottom" className="mx-auto max-w-[430px] rounded-t-2xl px-6 pb-8 pt-4">
   225→   225→          <SheetHeader>
   226→   226→            <SheetTitle className="text-xl font-bold text-gray-900">Пополнить баланс</SheetTitle>

   228→   228→          </SheetHeader>
   229→   229→          <div className="mt-4">

   233→   233→                <button
   234→   234→                  key={a}
   235→   235→                  type="button"
   236→   236→                  onClick={() => setTopUpAmount(String(a))}
   237→   237→                  className={`rounded-xl border-2 py-3 text-center text-lg font-semibold transition-all ${
   238→   238→                    topUpAmount === String(a)
   239→   239→                      ? 'border-orange-500 bg-orange-50 text-orange-600'
   240→   240→                      : 'border-gray-200 bg-white text-gray-700 hover:border-orange-300'
   241→   241→                  }`}
   242→   242→                >
   243→   243→                  {a} ₽
   244→   244→                </button>
   245→   245→              ))}
   246→   246→            </div>
   247→   247→            {/* Custom input */}

   262→   262→            {/* Submit */}
   263→   263→            <button

   266→   266→              disabled={topUpLoading || !topUpAmount}

   288→   288→
   289→   289→      {/* ============ NEWS DETAIL SHEET ============ */}
   290→   290→      <Sheet open={!!selectedNews} onOpenChange={() => setSelectedNews(null)}>

   296→   296→                <SheetDescription className="text-gray-400 text-sm">
   297→   297→                  <Clock className="mr-1 inline h-3.5 w-3.5" />
   298→   298→                  {formatDate(selectedNews.publishedAt)}
   299→   299→                  {selectedNews.readCount != null && ` · ${selectedNews.readCount} просмотров`}
   300→   300→                </SheetDescription>
   301→   301→              </SheetHeader>
   302→   302→              <p className="mt-4 text-[16px] leading-relaxed text-gray-700">{selectedNews.summary}</p>
   303→   303→            </>

   311→   311→          {tabs.map((tab, idx) => {

   313→   313→            const active = idx === activeTab;
   314→   314→            return (

   316→   316→                <button
   317→   317→                  type="button"

   319→   319→                  className={`flex flex-col items-center gap-1.5 transition-colors ${

   324→   324→                  <span className="text-[13px] font-medium">{tab.label}</span>

   326→   326→              </li>
